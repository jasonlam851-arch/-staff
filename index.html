window.triggerNext = function(g) { const c = currentData[g]?.currentCalled || 0; const t = currentData[g]?.totalIssued || 0; if (c < t) { // 同步寫入新的被叫號碼與獨一無二時間戳觸發語音 set(ref(db, `queueSystem/${g}`), { currentCalled: c + 1, totalIssued: t, lastCallTimestamp: Date.now() + "_" + Math.random() }); } else { alert(`${g}組候位名單已空。`); } }; window.triggerRecall = function(g) { const c = currentData[g]?.currentCalled || 0; const t = currentData[g]?.totalIssued || 0; if (c > 0) { // 號碼不變，僅改變隨機值戳記，強制大屏幕重新高呼 set(ref(db, `queueSystem/${g}`), { currentCalled: c, totalIssued: t, lastCallTimestamp: Date.now() + "_" + Math.random() }); } };
